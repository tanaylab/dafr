```r
# Assert that an expression does not call `as.matrix` on any object.
# Used to verify sparse fast paths do not densify their input.
assert_no_densify_during <- function(expr) {
    env <- parent.frame()
    calls <- 0L
    trace_fn <- function(...) {
        calls <<- calls + 1L
    }
    suppressMessages(
        trace(base::as.matrix, tracer = trace_fn, print = FALSE, where = baseenv())
    )
    on.exit(suppressMessages(untrace(base::as.matrix, where = baseenv())),
        add = TRUE)
    result <- eval(expr, envir = env)
    testthat::expect_equal(calls, 0L,
        info = "as.matrix was called during expression — sparse path densified"
    )
    result
}
```

**Purpose:** replaces the existing `matrixStats::rowMaxs(as.matrix(m))` densification at `R/query_eval.R:567–568, 582–583`. Smallest kernel; serves as template for Tasks 3–6.

**Files:**

- Modify: `src/cpp11.cpp` (regenerated — do not hand-edit; `cpp11::cpp_register()` does it)

- Test: `tests/testthat/test-kernels-slice8.R`

**Algorithm:** for axis == 1 (per-col output, ReduceToRow equivalent):
1. For each column j, compute `min(@x[pp[j]:pp[j+1]-1])` and `max(...)`.
2. If `pp[j+1] - pp[j] < nrow`, fold in `0` (implicit zeros exist).
3. If column is all-zero (`pp[j+1] == pp[j]` and `nrow > 0`), min = max = 0.

For axis == 0 (per-row output): use the per-thread row-accumulator pattern from `kernel_log_reduce_csc.cpp` lines 88–125 (initialise `pout[r]` to `+Inf` for Min / `-Inf` for Max; first-pass walks nnz and updates via thread-local buffers; second-pass folds in 0 for rows with `nnz_per_row[r] < ncol`).

```cpp
// src/kernel_minmax_csc.cpp
// Per-column or per-row min/max over a dgCMatrix, honouring implicit zeros.
// Template: src/kernel_log_reduce.cpp (same thread-bucket pattern for axis=0).
#include <cpp11.hpp>
#include "openmp_shim.h"
#include <algorithm>
#include <limits>
#include <string>
#include <vector>

[[cpp11::register]]
cpp11::writable::doubles kernel_minmax_csc_cpp(
    cpp11::doubles  x,
    cpp11::integers i,
    cpp11::integers p,
    int nrow,
    int ncol,
    int axis,                   // 0 = per-row (collapse cols), 1 = per-col
    std::string variant,        // "Min" | "Max"
    int threshold
) {
    const double *px = REAL(x.data());
    const int *pi = INTEGER(i.data());
    const int *pp = INTEGER(p.data());
    const bool is_min = (variant == "Min");
    const double sentinel = is_min
        ? std::numeric_limits<double>::infinity()
        : -std::numeric_limits<double>::infinity();
    auto fold = [&](double a, double b) { return is_min ? std::min(a, b) : std::max(a, b); };

    if (axis == 1) {
        cpp11::writable::doubles out(ncol);
        double *pout = REAL(out.data());
        DAFR_PARALLEL_FOR(ncol >= threshold)
        for (int j = 0; j < ncol; ++j) {
            const int nnz_j = pp[j + 1] - pp[j];
            double v = sentinel;
            for (int k = pp[j]; k < pp[j + 1]; ++k) v = fold(v, px[k]);
            if (nnz_j < nrow) v = fold(v, 0.0);
            if (nnz_j == 0 && nrow == 0) v = 0.0;   // empty column edge case
            pout[j] = v;
        }
        return out;
    }

    // axis == 0: per-row output. Thread-local buckets pattern from
    // kernel_log_reduce_csc_cpp (see src/kernel_log_reduce.cpp:88–125).
    cpp11::writable::doubles out(nrow);
    double *pout = REAL(out.data());
    for (int r = 0; r < nrow; ++r) pout[r] = sentinel;
    std::vector<int> nnz_per_row(nrow, 0);

    const int nthreads = dafr_omp_get_max_threads_capped(ncol, threshold);
    std::vector<std::vector<double>> tbuf(nthreads,
        std::vector<double>(nrow, sentinel));
    std::vector<std::vector<int>> tnnz(nthreads, std::vector<int>(nrow, 0));

    DAFR_PARALLEL_FOR(ncol >= threshold)
    for (int j = 0; j < ncol; ++j) {
        const int tid = dafr_omp_get_thread_num();
        auto &tb = tbuf[tid];
        auto &tn = tnnz[tid];
        for (int k = pp[j]; k < pp[j + 1]; ++k) {
            const int r = pi[k];
            tb[r] = fold(tb[r], px[k]);
            tn[r] += 1;
        }
    }
    for (int t = 0; t < nthreads; ++t) {
        for (int r = 0; r < nrow; ++r) {
            pout[r] = fold(pout[r], tbuf[t][r]);
            nnz_per_row[r] += tnnz[t][r];
        }
    }
    DAFR_PARALLEL_FOR(nrow >= threshold)
    for (int r = 0; r < nrow; ++r) {
        if (nnz_per_row[r] < ncol) pout[r] = fold(pout[r], 0.0);
    }
    return out;
}
```

## Task 3: `kernel_var_csc_cpp` + Var/Std/VarN/StdN wire-up

**Purpose:** one-pass Σx / Σx² / n_explicit per output slot; emits variance, std, VarN, StdN from shared statistics. Removes the densifying slow-path apply for all four variants.

**Files:**
- Create: `src/kernel_var_csc.cpp`
- Modify: `R/query_eval.R` — extend `.apply_reduction_fast` to dispatch Var/Std/VarN/StdN (this is where the params gate needs to change — see Step 5)
- Test: `tests/testthat/test-kernels-slice8.R`

**Algorithm:** one-pass CSC walk. For the per-col path (axis=1): `sum_x = Σ @x[k]`, `sum_x2 = Σ @x[k]²`, `nnz_j = pp[j+1] - pp[j]`. Then `mean = sum_x / nrow`, `var = (sum_x2 - nrow * mean²) / nrow` (uncorrected — matches Julia DAF `var!`). `VarN = var / (mean + eps)²`, `StdN = sqrt(var) / (mean + eps)`. For the per-row path use the thread-bucket pattern (both `sum_x` and `sum_x2` per row).

**Numerical note:** for sparse matrices with ~5% density, the implicit zeros contribute zero to both `sum_x` and `sum_x2`, so the one-pass formula is stable enough. If a kernel regression test fails on high-mean low-variance input, fall back to Welford in a follow-up — do not land the fallback unless the test fails.

```cpp
// src/kernel_var_csc.cpp
// One-pass variance / std / VarN / StdN over dgCMatrix columns (axis=1) or
// rows (axis=0). Uses sum_x + sum_x2 + counts pattern; implicit zeros
// contribute 0 to sums but their count is nrow - nnz_in_col.
#include <cpp11.hpp>
#include "openmp_shim.h"
#include <cmath>
#include <string>
#include <vector>

namespace {
inline double derive(const std::string &variant, double var_u,
                     double mean_u, double eps) {
    if (variant == "Var")  return var_u;
    if (variant == "Std")  return std::sqrt(var_u);
    if (variant == "VarN") return var_u / ((mean_u + eps) * (mean_u + eps));
    if (variant == "StdN") return std::sqrt(var_u) / (mean_u + eps);
    return 0.0;
}
}

[[cpp11::register]]
cpp11::writable::doubles kernel_var_csc_cpp(
    cpp11::doubles x, cpp11::integers i, cpp11::integers p,
    int nrow, int ncol, int axis,
    std::string variant,   // "Var" | "Std" | "VarN" | "StdN"
    double eps,
    int threshold
) {
    const double *px = REAL(x.data());
    const int *pi = INTEGER(i.data());
    const int *pp = INTEGER(p.data());

    if (axis == 1) {
        cpp11::writable::doubles out(ncol);
        double *pout = REAL(out.data());
        DAFR_PARALLEL_FOR(ncol >= threshold)
        for (int j = 0; j < ncol; ++j) {
            double sx = 0.0, sxx = 0.0;
            for (int k = pp[j]; k < pp[j + 1]; ++k) {
                const double v = px[k];
                sx += v;
                sxx += v * v;
            }
            const double mean = nrow > 0 ? sx / nrow : 0.0;
            const double var_u = nrow > 0 ? (sxx / nrow - mean * mean) : 0.0;
            pout[j] = derive(variant, var_u < 0 ? 0.0 : var_u, mean, eps);

    std::vector<std::vector<double>> tsxx(nthreads, std::vector<double>(nrow, 0.0));

    for (int j = 0; j < ncol; ++j) {
        const int tid = dafr_omp_get_thread_num();
        auto &tsa = tsx[tid];
        auto &tsb = tsxx[tid];
        for (int k = pp[j]; k < pp[j + 1]; ++k) {
            const int r = pi[k];
            const double v = px[k];
            tsa[r] += v;
            tsb[r] += v * v;
        }
    }
    for (int t = 0; t < nthreads; ++t) {
        for (int r = 0; r < nrow; ++r) {
            sx_tot[r]  += tsx[t][r];
            sxx_tot[r] += tsxx[t][r];
        }
    }
    DAFR_PARALLEL_FOR(nrow >= threshold)
    for (int r = 0; r < nrow; ++r) {
        const double mean = ncol > 0 ? sx_tot[r] / ncol : 0.0;
        const double var_u = ncol > 0 ? (sxx_tot[r] / ncol - mean * mean) : 0.0;
        pout[r] = derive(variant, var_u < 0 ? 0.0 : var_u, mean, eps);
    }
    return out;
}
```

```cpp
// src/kernel_geomean_csc.cpp
// Geometric mean with eps regularisation over dgCMatrix. Mirrors
// kernel_log_reduce_csc_cpp (fused log+reduce) + post-exp - eps step.
#include <cpp11.hpp>
#include "openmp_shim.h"
#include <cmath>
#include <vector>

[[cpp11::register]]
cpp11::writable::doubles kernel_geomean_csc_cpp(
    cpp11::doubles x, cpp11::integers i, cpp11::integers p,
    int nrow, int ncol, int axis, double eps, int threshold
) {
    const double *px = REAL(x.data());
    const int *pi = INTEGER(i.data());
    const int *pp = INTEGER(p.data());
    const double log_eps = std::log(eps);

    if (axis == 1) {
        cpp11::writable::doubles out(ncol);
        double *pout = REAL(out.data());
        DAFR_PARALLEL_FOR(ncol >= threshold)
        for (int j = 0; j < ncol; ++j) {
            const int nnz_j = pp[j + 1] - pp[j];
            double s = 0.0;
            for (int k = pp[j]; k < pp[j + 1]; ++k) s += std::log(px[k] + eps);
            s += (nrow - nnz_j) * log_eps;
            pout[j] = nrow > 0 ? std::exp(s / nrow) - eps : 0.0;
        }
        return out;
    }

    // axis == 0: per-row thread buckets (pattern from log_reduce)
    cpp11::writable::doubles out(nrow);
    double *pout = REAL(out.data());
    std::vector<double> sum_tot(nrow, 0.0);
    std::vector<int> nnz_per_row(nrow, 0);
    const int nthreads = dafr_omp_get_max_threads_capped(ncol, threshold);
    std::vector<std::vector<double>> tsum(nthreads, std::vector<double>(nrow, 0.0));
    std::vector<std::vector<int>> tnnz(nthreads, std::vector<int>(nrow, 0));

    DAFR_PARALLEL_FOR(ncol >= threshold)
    for (int j = 0; j < ncol; ++j) {
        const int tid = dafr_omp_get_thread_num();
        auto &ts = tsum[tid];
        auto &tn = tnnz[tid];
        for (int k = pp[j]; k < pp[j + 1]; ++k) {
            ts[pi[k]] += std::log(px[k] + eps);
            tn[pi[k]] += 1;
        }
    }
    for (int t = 0; t < nthreads; ++t) {
        for (int r = 0; r < nrow; ++r) {
            sum_tot[r]      += tsum[t][r];
            nnz_per_row[r]  += tnnz[t][r];
        }
    }
    DAFR_PARALLEL_FOR(nrow >= threshold)
    for (int r = 0; r < nrow; ++r) {
        const double s = sum_tot[r] + (ncol - nnz_per_row[r]) * log_eps;
        pout[r] = ncol > 0 ? std::exp(s / ncol) - eps : 0.0;
    }
    return out;
}
```

## Task 5: `kernel_quantile_csc_cpp` + Median/Quantile wire-up

**Purpose:** per-column (or per-row) quantile at arbitrary `q ∈ [0, 1]`, sparsity-aware (implicit zeros count but aren't materialised).

**Files:**
- Create: `src/kernel_quantile_csc.cpp`
- Modify: `R/query_eval.R`
- Test: `tests/testthat/test-kernels-slice8.R`

**Algorithm (per-column, axis=1):**

1. `n = nrow`, `nnz_j = pp[j+1] - pp[j]`, `n_zeros = n - nnz_j`.
2. Type-7 rank: `h = q * (n - 1)`, `lo = floor(h)`, `hi = ceil(h)`, `frac = h - lo`.
3. Copy `@x[pp[j]:pp[j+1]-1]` into a local buffer; partition by sign (`neg` count, `pos` count).
4. The implicit zeros occupy ranks `[neg .. neg + n_zeros - 1]` in the sorted order.
5. For each rank `r ∈ {lo, hi}`:
   - If `r < neg`: pick the `(r+1)`-th smallest from `neg[]` via `std::nth_element`.
   - Else if `r < neg + n_zeros`: value is 0.
   - Else: pick the `(r - neg - n_zeros + 1)`-th smallest from `pos[]`.
6. Interpolate: `val = (1 - frac) * val_lo + frac * val_hi`.
7. Special-case q=0.5 (Median) = same algorithm, no new code.

Per-row (axis=0): not feasible to do CSC-column-parallel without per-row buffers. Use the thread-buckets pattern to collect per-row `@x` into `std::vector<double>[nrow]` first pass, then per-row quantile second pass (parallel over rows).

```cpp
// src/kernel_quantile_csc.cpp
// Per-column or per-row quantile at q (type-7) over dgCMatrix, treating
// implicit zeros as their own rank band without materialising them.
#include <cpp11.hpp>
#include "openmp_shim.h"
#include <algorithm>
#include <cmath>
#include <vector>

namespace {

// Pick the k-th smallest (0-based) from a partitioned neg/zeros/pos layout
// without materialising zeros.
double pick_rank(std::vector<double> &neg, std::vector<double> &pos,
                int n_zeros, int k) {
    const int n_neg = static_cast<int>(neg.size());
    if (k < n_neg) {
        std::nth_element(neg.begin(), neg.begin() + k, neg.end());
        return neg[k];
    }
    if (k < n_neg + n_zeros) return 0.0;
    const int k_pos = k - n_neg - n_zeros;
    std::nth_element(pos.begin(), pos.begin() + k_pos, pos.end());
    return pos[k_pos];
}

double column_quantile(const double *px, const int *pi, int start, int end,
                       int n_rows, double q) {
    (void)pi;
    const int nnz = end - start;
    const int n = n_rows;
    if (n == 0) return 0.0;
    std::vector<double> neg, pos;
    neg.reserve(nnz); pos.reserve(nnz);
    for (int k = start; k < end; ++k) {
        if (px[k] < 0) neg.push_back(px[k]);
        else if (px[k] > 0) pos.push_back(px[k]);
        // explicit zeros: treat as implicit zeros (same rank band)
    }
    const int n_zeros = n - (int)neg.size() - (int)pos.size();
    const double h = q * (n - 1);
    const int lo = (int)std::floor(h);
    const int hi = (int)std::ceil(h);
    const double frac = h - lo;
    double vlo, vhi;
    if (lo == hi) {
        return pick_rank(neg, pos, n_zeros, lo);
    }
    vlo = pick_rank(neg, pos, n_zeros, lo);
    // Recompute neg/pos because nth_element left them in partitioned order;
    // for the second pick_rank call on neg or pos we can reuse as-is because
    // we only use nth_element (partial sort), but to avoid subtle bugs we
    // rebuild. nnz << nrow on sparse inputs so cost is fine.
    neg.clear(); pos.clear();
    for (int k = start; k < end; ++k) {
        if (px[k] < 0) neg.push_back(px[k]);
        else if (px[k] > 0) pos.push_back(px[k]);
    }
    vhi = pick_rank(neg, pos, n_zeros, hi);
    return (1 - frac) * vlo + frac * vhi;
}

} // anonymous namespace

[[cpp11::register]]
cpp11::writable::doubles kernel_quantile_csc_cpp(
    cpp11::doubles x, cpp11::integers i, cpp11::integers p,
    int nrow, int ncol, int axis, double q, int threshold
) {
    const double *px = REAL(x.data());
    const int *pi = INTEGER(i.data());
    const int *pp = INTEGER(p.data());

    if (axis == 1) {
        cpp11::writable::doubles out(ncol);
        double *pout = REAL(out.data());
        DAFR_PARALLEL_FOR(ncol >= threshold)
        for (int j = 0; j < ncol; ++j) {
            pout[j] = column_quantile(px, pi, pp[j], pp[j + 1], nrow, q);
        }
        return out;
    }

    // axis == 0: collect per-row values first, then per-row nth_element.
    std::vector<std::vector<double>> rows(nrow);
    for (int j = 0; j < ncol; ++j) {
        for (int k = pp[j]; k < pp[j + 1]; ++k) {
            rows[pi[k]].push_back(px[k]);
        }
    }
    cpp11::writable::doubles out(nrow);
    double *pout = REAL(out.data());
    DAFR_PARALLEL_FOR(nrow >= threshold)
    for (int r = 0; r < nrow; ++r) {
        std::vector<double> neg, pos;
        neg.reserve(rows[r].size());
        pos.reserve(rows[r].size());
        for (double v : rows[r]) {
            if (v < 0) neg.push_back(v);
            else if (v > 0) pos.push_back(v);
        }
        const int n = ncol;
        const int n_zeros = n - (int)neg.size() - (int)pos.size();
        const double h = q * (n - 1);
        const int lo = (int)std::floor(h);
        const int hi = (int)std::ceil(h);
        const double frac = h - lo;
        if (lo == hi) {
            pout[r] = pick_rank(neg, pos, n_zeros, lo);
        } else {
            const double v_lo = pick_rank(neg, pos, n_zeros, lo);
            // rebuild to reuse neg/pos for hi pick
            neg.clear(); pos.clear();
            for (double v : rows[r]) {
                if (v < 0) neg.push_back(v);
                else if (v > 0) pos.push_back(v);
            }
            const double v_hi = pick_rank(neg, pos, n_zeros, hi);
            pout[r] = (1 - frac) * v_lo + frac * v_hi;
        }
    }
    return out;
}
```

```cpp
// src/kernel_mode_csc.cpp
#include <cpp11.hpp>
#include "openmp_shim.h"
#include <unordered_map>
#include <vector>

namespace {
double column_mode(const double *px, int start, int end, int n_rows) {
    std::unordered_map<double, int> counts;
    counts.reserve(end - start);
    for (int k = start; k < end; ++k) counts[px[k]] += 1;
    const int n_zeros = n_rows - (end - start);
    double best_val = 0.0;
    int best_count = n_zeros;
    for (auto &kv : counts) {
        if (kv.second > best_count ||
            (kv.second == best_count && kv.first < best_val)) {
            best_count = kv.second;
            best_val = kv.first;
        }
    }
    return best_val;
}

[[cpp11::register]]
cpp11::writable::doubles kernel_mode_csc_cpp(
    cpp11::doubles x, cpp11::integers i, cpp11::integers p,
    int nrow, int ncol, int axis, int threshold
) {
    const double *px = REAL(x.data());
    const int *pi = INTEGER(i.data());
    const int *pp = INTEGER(p.data());

    if (axis == 1) {
        cpp11::writable::doubles out(ncol);
        double *pout = REAL(out.data());
        DAFR_PARALLEL_FOR(ncol >= threshold)
        for (int j = 0; j < ncol; ++j) {
            pout[j] = column_mode(px, pp[j], pp[j + 1], nrow);
        }
        return out;
    }

    // axis == 0: per-row. Collect then per-row hash.
    std::vector<std::vector<double>> rows(nrow);
    for (int j = 0; j < ncol; ++j) {
        for (int k = pp[j]; k < pp[j + 1]; ++k) {
            rows[pi[k]].push_back(px[k]);
        }
    }
    cpp11::writable::doubles out(nrow);
    double *pout = REAL(out.data());
    DAFR_PARALLEL_FOR(nrow >= threshold)
    for (int r = 0; r < nrow; ++r) {
        std::unordered_map<double, int> counts;
        for (double v : rows[r]) counts[v] += 1;
        const int n_zeros = ncol - (int)rows[r].size();
        double best_val = 0.0;
        int best_count = n_zeros;
        for (auto &kv : counts) {
            if (kv.second > best_count ||
                (kv.second == best_count && kv.first < best_val)) {
                best_count = kv.second;
                best_val = kv.first;
            }
        }
        pout[r] = best_val;
    }
    return out;
}
```

## Task 8: `kernel_grouped_reduce_csc_cpp` + `kernel_grouped_reduce_dense_cpp`

**Purpose:** one shared engine for patterns G2 (row-group, ReduceToColumn → `ngroups × ncol`) and G3 (col-group, ReduceToRow → `nrow × ngroups`) for the op family Sum/Mean/Min/Max/Var/Std/VarN/StdN/GeoMean. Median/Quantile and Mode get separate kernels in Task 9.

**Files:**
- Create: `src/kernel_grouped_reduce_csc.cpp`
- Create: `src/kernel_grouped_reduce_dense.cpp`
- Test: `tests/testthat/test-kernels-slice8.R`

**Algorithm (sparse CSC, G2 = row-group):**

Output is `ngroups × ncol` (dense). For each column j in parallel:
1. Local per-group accumulators: `sum_x[g]`, `sum_x2[g]`, `nnz_per_group[g]`, `min_x[g]`, `max_x[g]`, `log_sum[g]`.
2. Iterate `k ∈ [pp[j], pp[j+1])`: `g = group[pi[k]] - 1`, update accumulators.
3. For each group g, compute total row count in that group from `n_in_group[g]` (precomputed once). `n_zeros_in_group = n_in_group[g] - nnz_per_group[g]`.
4. Derive per-op output: Sum = `sum_x[g]`; Mean = `sum_x[g] / n_in_group[g]`; Min = `min_x[g]` folded with 0 if `n_zeros > 0`; Var from `sum_x`/`sum_x2` as in Task 3; GeoMean from `log_sum + n_zeros * log(eps)` as in Task 4.
5. Write to `out[g, j]` (column-major).

**Algorithm (sparse CSC, G3 = col-group):**

Output is `nrow × ngroups` (dense). Thread-bucket pattern by (row, group): for each j in parallel, for each nnz in column, update `tbuf[tid][pi[k] + group[j] * nrow]`. Post-merge thread buffers, post-process per op.

**Algorithm (dense, both axes):** `rowsum` (base R, already C) for Sum/Mean. For Var/Std/Min/Max/etc, a single dense kernel that loops over `(group, col)` or `(row, group)` cells.

The kernel signature is shared (sparse and dense):

```cpp
// src/kernel_grouped_reduce_csc.cpp
// Single-engine grouped reduction for ops: Sum, Mean, Min, Max, Var, Std,
// VarN, StdN, GeoMean over dgCMatrix. Two axes:
//   axis == 2 (G2): group is per-row index; output is ngroups x ncol.
//   axis == 3 (G3): group is per-col index; output is nrow x ngroups.
// Median/Quantile/Mode grouped variants are in separate kernels (Task 9).
#include <cpp11.hpp>
#include "openmp_shim.h"
#include <algorithm>
#include <cmath>
#include <limits>
#include <string>
#include <vector>

// Shared per-op accumulator bundle. Minimal but sufficient for the 9 ops.
struct Acc {
    double sum_x = 0.0;
    double sum_x2 = 0.0;
    double log_sum = 0.0;                    // Σ log(x + eps) over explicit entries
    int nnz = 0;
    double min_x =  std::numeric_limits<double>::infinity();
    double max_x = -std::numeric_limits<double>::infinity();
    void push(double v, double eps) {
        sum_x += v; sum_x2 += v * v;
        log_sum += std::log(v + eps);
        nnz += 1;
        if (v < min_x) min_x = v;
        if (v > max_x) max_x = v;
    }
};

inline double derive_op(const std::string &op, const Acc &a,
                        int n_total, double eps) {
    const int n_zeros = n_total - a.nnz;
    const double sum_total = a.sum_x;  // zeros add 0
    const double mean = n_total > 0 ? sum_total / n_total : 0.0;
    const double var_u = n_total > 0
        ? ((a.sum_x2 + /*zeros contribute 0*/ 0.0) / n_total - mean * mean)
        : 0.0;
    const double var_u_pos = var_u < 0 ? 0.0 : var_u;
    if (op == "Sum")    return sum_total;
    if (op == "Mean")   return mean;
    if (op == "Min") {
        if (a.nnz == 0) return n_total > 0 ? 0.0
            : std::numeric_limits<double>::infinity();
        return n_zeros > 0 ? std::min(a.min_x, 0.0) : a.min_x;
    }
    if (op == "Max") {
        if (a.nnz == 0) return n_total > 0 ? 0.0
            : -std::numeric_limits<double>::infinity();
        return n_zeros > 0 ? std::max(a.max_x, 0.0) : a.max_x;
    }
    if (op == "Var")    return var_u_pos;
    if (op == "Std")    return std::sqrt(var_u_pos);
    // NOTE: VarN denominator is LINEAR per .op_varn in R/operations.R,
    // NOT squared. The original skeleton had (mean+eps)^2 which is wrong.
    if (op == "VarN")   return var_u_pos / (mean + eps);
    if (op == "StdN")   return std::sqrt(var_u_pos) / (mean + eps);
    if (op == "GeoMean") {
        const double s = a.log_sum + n_zeros * std::log(eps);
        return n_total > 0 ? std::exp(s / n_total) - eps : 0.0;
    }
    return 0.0;
}

[[cpp11::register]]
cpp11::writable::doubles_matrix<cpp11::by_column> kernel_grouped_reduce_csc_cpp(
    cpp11::doubles x, cpp11::integers i, cpp11::integers p,
    int nrow, int ncol,
    cpp11::integers group, int ngroups, cpp11::integers n_in_group,
    int axis, std::string op, double eps, int threshold
) {
    const double *px = REAL(x.data());
    const int *pi = INTEGER(i.data());
    const int *pp = INTEGER(p.data());
    const int *pg = INTEGER(group.data());
    const int *png = INTEGER(n_in_group.data());

    if (axis == 2) {
        // G2: per-col accumulators are per-group.
        cpp11::writable::doubles_matrix<cpp11::by_column> out(ngroups, ncol);
        DAFR_PARALLEL_FOR(ncol >= threshold)
        for (int j = 0; j < ncol; ++j) {
            std::vector<Acc> accs(ngroups);
            for (int k = pp[j]; k < pp[j + 1]; ++k) {
                accs[pg[pi[k]] - 1].push(px[k], eps);
            }
            for (int g = 0; g < ngroups; ++g) {
                out(g, j) = derive_op(op, accs[g], png[g], eps);
            }
        }
        return out;
    }

    // axis == 3: G3 — output nrow x ngroups. Accumulate per-(row, group)
    // via thread buckets (nrow * ngroups per thread).
    cpp11::writable::doubles_matrix<cpp11::by_column> out(nrow, ngroups);
    const int nthreads = dafr_omp_get_max_threads_capped(ncol, threshold);
    std::vector<std::vector<Acc>> tacc(nthreads,
        std::vector<Acc>((size_t)nrow * ngroups));

    DAFR_PARALLEL_FOR(ncol >= threshold)
    for (int j = 0; j < ncol; ++j) {
        const int tid = dafr_omp_get_thread_num();
        const int g = pg[j] - 1;
        auto &tbuf = tacc[tid];
        for (int k = pp[j]; k < pp[j + 1]; ++k) {
            tbuf[(size_t)pi[k] + (size_t)g * nrow].push(px[k], eps);
        }
    }
    std::vector<Acc> accs((size_t)nrow * ngroups);
    for (int t = 0; t < nthreads; ++t) {
        auto &tb = tacc[t];
        for (size_t idx = 0; idx < accs.size(); ++idx) {
            accs[idx].sum_x   += tb[idx].sum_x;
            accs[idx].sum_x2  += tb[idx].sum_x2;
            accs[idx].log_sum += tb[idx].log_sum;
            accs[idx].nnz     += tb[idx].nnz;
            accs[idx].min_x    = std::min(accs[idx].min_x, tb[idx].min_x);
            accs[idx].max_x    = std::max(accs[idx].max_x, tb[idx].max_x);
        }
    }
    DAFR_PARALLEL_FOR(nrow >= threshold)
    for (int r = 0; r < nrow; ++r) {
        for (int g = 0; g < ngroups; ++g) {
            out(r, g) = derive_op(op, accs[(size_t)r + (size_t)g * nrow],
                                   png[g], eps);
        }
    }
    return out;
}
```

```r
.apply_reduction_grouped_vector <- function(node, state, daf) {
    fn <- get_reduction(node$reduction)
    params <- .coerce_params(node$params)
    x <- state$value
    g <- state$pending_groups
    ngroups <- length(unique(g))  # preserve group order via factor levels
    # Use factor to stabilise group ordering (levels in first-appearance order)
    gfac <- factor(g, levels = unique(g))
    gi <- as.integer(gfac)

    label <- .reduction_builtin_label(fn)

    # Mode-on-character: dedicated R helper
    if (!is.na(label) && label == "Mode" && is.character(x)) {
        vals <- .grouped_mode_character(x, gi, nlevels(gfac))
        return(list(kind = "vector", axis = NULL,
                    value = stats::setNames(vals, levels(gfac))))
    }

    # Numeric builtin fast path
    if (!is.na(label) && is.numeric(x)) {
        vals <- .grouped_vector_builtin(x, gi, nlevels(gfac), label, params)
        return(list(kind = "vector", axis = NULL,
                    value = stats::setNames(vals, levels(gfac))))
    }

    # Fallback: type-sniffed split + vectorised loop (no numeric(1) cast)
    idx <- split(seq_along(x), gi)
    sample_idx <- idx[[which(lengths(idx) > 0L)[1L]]]
    mode <- .grouped_proto_storage(fn, x[sample_idx], params)
    out <- .grouped_allocate_output(mode, nlevels(gfac))
    for (g_i in seq_along(idx)) {
        if (length(idx[[g_i]]) == 0L) { out[g_i] <- NA; next }
        out[g_i] <- do.call(fn, c(list(x[idx[[g_i]]]), params))
    }
    list(kind = "vector", axis = NULL,
         value = stats::setNames(out, levels(gfac)))
```

```r
.grouped_vector_builtin <- function(x, group, ngroups, label, params) {
    n_in_group <- as.integer(tabulate(group, ngroups))
    if (label %in% c("Sum","Mean","Min","Max","Var","Std","VarN","StdN","GeoMean")) {
        # Reuse the grouped matrix kernel by treating x as a 1-row "matrix"
        # OR — simpler — implement a thin 1D variant via rowsum + derivations.
        sum_x  <- as.numeric(rowsum(x,       group))   # length ngroups
        sum_x2 <- as.numeric(rowsum(x * x,   group))
        means  <- sum_x / n_in_group
        vars   <- pmax(sum_x2 / n_in_group - means^2, 0)
        eps    <- if (label %in% c("VarN","StdN")) .param_eps(params)
                   else if (label == "GeoMean") .param_eps(params)
                   else 0
        switch(label,
            Sum     = sum_x,
            Mean    = means,
            Min     = as.numeric(tapply(x, group, min)),
            Max     = as.numeric(tapply(x, group, max)),
            Var     = vars,
            Std     = sqrt(vars),
            VarN    = vars / (means + eps)^2,
            StdN    = sqrt(vars) / (means + eps),
            GeoMean = {
                log_sum <- as.numeric(rowsum(log(x + eps), group))
                exp(log_sum / n_in_group) - eps
            })
    } else if (label == "Median" || label == "Quantile") {
        q <- if (label == "Median") 0.5 else .param_quantile_q(params)
        vapply(split(x, group), function(v) stats::quantile(v, q, type = 7,
            names = FALSE), numeric(1))
    } else if (label == "Mode") {
        vapply(split(x, group), function(v) {
            tt <- table(v)
            as.numeric(names(tt)[which.max(tt)])
        }, numeric(1))
    } else stop("unreachable")
}

.grouped_mode_character <- function(x, group, ngroups) {
    out <- character(ngroups)
    for (g_i in seq_len(ngroups)) {
        vals <- x[group == g_i]
        if (length(vals) == 0L) { out[g_i] <- NA_character_; next }
        tt <- table(vals)
        out[g_i] <- names(tt)[which.max(tt)]
    }
    out
}
```

```r
.apply_reduction_grouped_matrix <- function(node, state, daf, by) {
    fn <- get_reduction(node$reduction)
    params <- .coerce_params(node$params)
    m <- state$value
    label <- .reduction_builtin_label(fn)

    is_g2 <- identical(by, "rows") && identical(node$op, "ReduceToColumn")
    is_g3 <- identical(by, "cols") && identical(node$op, "ReduceToRow")
    is_g4 <- (identical(by, "rows") && identical(node$op, "ReduceToRow")) ||
             (identical(by, "cols") && identical(node$op, "ReduceToColumn"))

    if (is_g4) {
        # Decompose: inner reduce-other-axis -> vector, outer grouped-vector
        inner <- .apply_reduction(
            list(op = if (identical(by, "rows")) "ReduceToRow" else "ReduceToColumn",
                 reduction = node$reduction, params = node$params),
            state = list(kind = "matrix", value = m,
                         rows_axis = state$rows_axis,
                         cols_axis = state$cols_axis),
            daf)
        # now inner is a vector state — graft on the pending group and recurse
        new_state <- list(kind = "grouped_vector",
            value = inner$value,
            pending_groups = if (identical(by, "rows")) state$pending_row_groups
                             else state$pending_col_groups)
        return(.apply_reduction_grouped_vector(node, new_state, daf))
    }

    if (is_g2 || is_g3) {
        g <- if (is_g2) state$pending_row_groups else state$pending_col_groups
        gfac <- factor(g, levels = unique(g))
        gi <- as.integer(gfac)
        ngroups <- nlevels(gfac)
        n_in_group <- as.integer(tabulate(gi, ngroups))

        # Numeric builtin single-reduction path
        if (!is.na(label)) {
            axis <- if (is_g2) 2L else 3L
            out <- .grouped_matrix_builtin(m, gi, ngroups, n_in_group,
                axis, label, params)
            rn <- if (is_g2) levels(gfac) else rownames(m)
            cn <- if (is_g2) colnames(m) else levels(gfac)
            return(list(
                kind = "matrix",
                rows_axis = if (is_g2) NULL else state$rows_axis,
                cols_axis = if (is_g2) state$cols_axis else NULL,
                value = { dimnames(out) <- list(rn, cn); out }))
        }

        # Fallback: split+apply with type-sniffed output
        return(.apply_reduction_grouped_matrix_fallback(m, gi, ngroups,
            node, fn, params, by, state))
    }

    stop(sprintf("unsupported grouped matrix pattern: op=%s by=%s",
        node$op, by), call. = FALSE)
}

.grouped_matrix_builtin <- function(m, gi, ngroups, n_in_group,
```

```r
.op_convert <- function(x, ..., type) {
    if (missing(type)) {
        stop("Convert: 'type' parameter is required (one of 'double', 'integer', 'logical')",
            call. = FALSE)
    }
    if (!is.character(type) || length(type) != 1L ||
        !type %in% c("double", "integer", "logical")) {
        stop(sprintf(
            "Convert: 'type' must be one of 'double', 'integer', 'logical' (got %s)",
            sQuote(as.character(type)[1L])
        ), call. = FALSE)
    }
    # Sparse preservation
    if (methods::is(x, "dgCMatrix")) {
        if (type == "double") return(x)
        if (type == "integer") {
            if (length(x@x) > 0L && any(x@x != floor(x@x))) {
                stop("Convert: non-integer value in integer coercion",
                    call. = FALSE)
            }
            x@x <- as.double(as.integer(x@x))
            return(x)
        }
        if (type == "logical") {
            x@x <- as.double(x@x != 0)
            return(Matrix::drop0(x))
        }
    }
    # Dense or vector — existing behaviour
    storage.mode(x) <- type
    x
}
```

```r
complete_daf <- function(leaf, mode = "r", name = NULL) {
    if (!mode %in% c("r", "r+")) {
        stop("`mode` must be \"r\" or \"r+\"", call. = FALSE)
    }
    stack <- list()
    path <- leaf
    while (!is.null(path)) {
        is_leaf <- length(stack) == 0L
        open_mode <- if (is_leaf) mode else "r"
        d <- open_daf(path, open_mode)
        stack <- c(stack, list(d))
        next_path <- if (format_has_scalar(d, "base_daf_repository")) {
            base <- format_get_scalar(d, "base_daf_repository")
            if (!startsWith(base, "/")) {
                base <- normalizePath(file.path(dirname(path), base),
                                      mustWork = FALSE)
            }
            base
        } else NULL
        path <- next_path
    }
    # `stack` is leaf-first; the leaf holds the optional base_daf_view
    # JSON. Remember the leaf before we rev() for chain construction.
    leaf_daf <- stack[[1L]]
    readers <- rev(stack)
    chain_name <- name %||% basename(leaf)
    chain <- if (length(readers) == 1L) {
        readers[[1L]]
    } else if (identical(mode, "r")) {
        chain_reader(readers, name = chain_name)
    } else {
        chain_writer(readers, name = chain_name)
    }
    if (format_has_scalar(leaf_daf, "base_daf_view")) {
        spec <- jsonlite::fromJSON(
            format_get_scalar(leaf_daf, "base_daf_view"),
            simplifyVector = FALSE
        )
        chain <- viewer(chain, name = chain_name,
            axes = spec$axes %||% list(),
            data = spec$data %||% list())
    }
    chain
}
```

```r
run_gate <- function(name, g) {
    cat(sprintf("[gate] %s ... ", name))
    d <- g$setup()
    bm <- bench::mark(
        baseline = g$baseline(d),
        fast = g$fast(d),
        iterations = 3L, check = FALSE, filter_gc = FALSE
    )
    t_base <- as.numeric(bm$median[bm$expression == "baseline"])
    t_fast <- as.numeric(bm$median[bm$expression == "fast"])
    ratio <- t_base / t_fast
    mem_base <- as.numeric(bm$mem_alloc[bm$expression == "baseline"])
    mem_fast <- as.numeric(bm$mem_alloc[bm$expression == "fast"])
    mem_ratio <- if (mem_fast > 0) mem_base / mem_fast else NA_real_
    status <- if (ratio >= g$ratio_target &&
                  (is.na(g$mem_ratio_target) || mem_ratio >= g$mem_ratio_target)) "PASS"
              else "FAIL"
    cat(sprintf("ratio=%.2fx (target >=%.1fx), mem_ratio=%.2fx (target %s): %s\n",
        ratio, g$ratio_target, mem_ratio,
        if (is.na(g$mem_ratio_target)) "—"
        else sprintf(">=%.1fx", g$mem_ratio_target), status))
    list(name = name, ratio = ratio, mem_ratio = mem_ratio,
         target = g$ratio_target, mem_target = g$mem_ratio_target,
         status = status)
}
```
